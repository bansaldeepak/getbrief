# S-008-T1: Write item.test.mjs — Test Existing CRUD Engine

**Priority:** P0 | **Status:** pending | **Blocked By:** --

## What

Write tests for `brief/lib/commands/item.mjs` — the generic CRUD engine that handles todo, decision, research, and audit commands. This has ZERO test coverage today. Since bug.mjs and memory.mjs will delegate to it (T2, T3), we need confidence it works correctly before building on it.

## Why

item.mjs is a 366-line module with zero tests. Bug and memory commands will delegate ALL their CRUD operations to it. If item.mjs has latent bugs, they become bug.mjs/memory.mjs bugs too. Testing now is cheaper than debugging delegation failures later.

## Context

**item.mjs exports one function:**
```javascript
export async function itemCommand(root, type, action, args, rawArgs)
```

**Actions:** `list`, `create`, `update`, `show`, `help`

**Key internals:**
- `FILE_DIRS` — maps type → directory (`todo: "documentation/todo"`, etc.)
- `TEMPLATES` — maps type → markdown template function
- `getItems(root)` — reads tracker.yaml, falls back to `scanFiles(root)`
- Uses `readTracker`, `writeTracker`, `nextId` from `../tracker.mjs`
- Uses `readBody` from `../body.mjs` (supports `--body=` flag and stdin pipe)
- Uses `table`, `statusOrder` from `../format.mjs`

**Existing test infrastructure:**
- Test runner: `node:test` (Node.js built-in)
- Existing tests: `brief/test/init.test.mjs`, `brief/test/story.test.mjs`, `brief/test/tracker.test.mjs`, `brief/test/context.test.mjs`
- Pattern: Create temp dir, set up fixture files, call functions, assert results

**Step 2 — Test `create` action:**
```javascript
import { test, describe, beforeEach, afterEach } from "node:test";
import assert from "node:assert/strict";
import { mkdtempSync, mkdirSync, writeFileSync, readFileSync, existsSync } from "node:fs";
import { join } from "node:path";
import { tmpdir } from "node:os";
import { rmSync } from "node:fs";
import { itemCommand } from "../lib/commands/item.mjs";

describe("itemCommand", () => {
  let root;

  beforeEach(() => {
    root = mkdtempSync(join(tmpdir(), "brief-item-test-"));
    mkdirSync(join(root, "documentation"), { recursive: true });
    // Seed empty tracker
    writeFileSync(join(root, "documentation/tracker.yaml"), "items: []\n");
  });

  afterEach(() => {
    rmSync(root, { recursive: true, force: true });
  });

  describe("create", () => {
    test("creates todo with correct ID and file", async () => {
      await itemCommand(root, "todo", "create", ["Fix the widget"], []);
      // Verify tracker entry
      const tracker = readFileSync(join(root, "documentation/tracker.yaml"), "utf8");
      assert.match(tracker, /TODO-001/);
      assert.match(tracker, /Fix the widget/);
      // Verify markdown file created
      assert.ok(existsSync(join(root, "documentation/todo/TODO-001.md")));
    });

    test("creates decision with correct prefix", async () => {
      await itemCommand(root, "decision", "create", ["Use PostgreSQL"], []);
      const tracker = readFileSync(join(root, "documentation/tracker.yaml"), "utf8");
      assert.match(tracker, /DEC-001/);
    });

    test("creates with --body flag", async () => {
      await itemCommand(root, "todo", "create", ["Title here"], ["brief", "todo", "create", "Title here", "--body=Detailed body"]);
      const content = readFileSync(join(root, "documentation/todo/TODO-001.md"), "utf8");
      assert.match(content, /Detailed body/);
    });

    test("increments ID correctly", async () => {
      await itemCommand(root, "todo", "create", ["First"], []);
      await itemCommand(root, "todo", "create", ["Second"], []);
      const tracker = readFileSync(join(root, "documentation/tracker.yaml"), "utf8");
      assert.match(tracker, /TODO-002/);
    });
  });
});
```
