**Priority:** P0 | **Status:** pending | **Blocked By:** S-012-T3

## What
Create the new `lib/harness.mjs` module that owns HARNESS copy and marker injection. Create `test/harness.test.mjs` with unit tests for all 5 injection matrix rows plus error paths. Module exports two functions: `detectHarnessState()` and `applyHarness()`.

## Why
This is the core new module for the feature. The detect/apply split (eng review D1) allows the init flow to show planned actions before the user confirms.

## Context

### Injection Matrix (from PRD)

| Row | AGENTS.md | CLAUDE.md | Action |
|-----|-----------|-----------|--------|
| 1 | missing | missing | Create both. AGENTS.md with skeleton + inline marker. CLAUDE.md with `@../AGENTS.md`. |
| 2 | exists | missing | Add inline marker to AGENTS.md. Create CLAUDE.md importing AGENTS.md. |
| 3 | exists | exists, imports AGENTS | Add inline marker to AGENTS.md only. |
| 4 | exists | exists, no import | Add inline marker to AGENTS.md. Add `@` import marker to CLAUDE.md. |
| 5 | missing | exists | Add `@` import marker to CLAUDE.md only. |

### Import Detection
CLAUDE.md imports AGENTS.md if it contains `@../AGENTS.md` or `@AGENTS.md`. Regex: `/@\.\.\/AGENTS\.md|@AGENTS\.md/`

### Marker Format
AGENTS.md (inline):
```markdown
<!-- brief:auto section="harness" -->
## Brief Workflow
<full trimmed HARNESS content>
<!-- /brief:auto -->
```

CLAUDE.md (import, only when no AGENTS.md import):
```markdown
<!-- brief:auto section="harness" -->
@../.brief/HARNESS.md
<!-- /brief:auto -->
```

### Existing Infrastructure
`lib/markers.mjs` exports:
- `parseMarkers(content)` — splits content into manual/auto sections
- `mergeContent(sections, newGenerated)` — idempotent merge, replaces existing auto sections

Atomic write pattern (from `writeManifest()` in init.mjs:268-270):
```js
const tmpPath = path + ".tmp";
writeFileSync(tmpPath, content);
renameSync(tmpPath, path);
```

Template resolution pattern (from init.mjs:12):
```js
const TEMPLATES_DIR = join(__dirname, "../templates/project-structure");
```

### AGENTS.md skeleton template
The template at `templates/project-structure/AGENTS.md` (47 lines) is the skeleton used when creating a new AGENTS.md. Read it before implementing `createAgentsWithHarness()`.

## How

**Step 1 — Create lib/harness.mjs:**
- File: `lib/harness.mjs`

```js
import { existsSync, readFileSync, writeFileSync, mkdirSync, renameSync } from "node:fs";
import { join, dirname } from "node:path";
import { fileURLToPath } from "node:url";
import { parseMarkers, mergeContent } from "./markers.mjs";

const __dirname = dirname(fileURLToPath(import.meta.url));
const HARNESS_TEMPLATE = join(__dirname, "../templates/HARNESS.md");
const AGENTS_TEMPLATE = join(__dirname, "../templates/project-structure/AGENTS.md");
const IMPORT_RE = /@\.\.\/AGENTS\.md|@AGENTS\.md/;

export function detectHarnessState(projectRoot) {
  // Verify template exists
  if (!existsSync(HARNESS_TEMPLATE)) {
    throw new Error(`HARNESS template not found at ${HARNESS_TEMPLATE}`);
  }

  const agentsMdPath = join(projectRoot, "AGENTS.md");
  const claudeMdPath = join(projectRoot, ".claude", "CLAUDE.md");

  const agentsExists = existsSync(agentsMdPath);
  const claudeExists = existsSync(claudeMdPath);

  let claudeImportsAgents = false;
  if (claudeExists) {
    const content = readFileSync(claudeMdPath, "utf8");
    claudeImportsAgents = IMPORT_RE.test(content);
  }

  // Determine matrix row
  let matrixRow;
  const actions = [];

  if (!agentsExists && !claudeExists) {
    matrixRow = 1;
    actions.push("Create AGENTS.md with skeleton + inline HARNESS");
    actions.push("Create CLAUDE.md importing AGENTS.md");
  } else if (agentsExists && !claudeExists) {
    matrixRow = 2;
    actions.push("Add inline HARNESS marker to AGENTS.md");
    actions.push("Create CLAUDE.md importing AGENTS.md");
  } else if (agentsExists && claudeExists && claudeImportsAgents) {
    matrixRow = 3;
    actions.push("Add inline HARNESS marker to AGENTS.md");
  } else if (agentsExists && claudeExists && !claudeImportsAgents) {
    matrixRow = 4;
    actions.push("Add inline HARNESS marker to AGENTS.md");
    actions.push("Add @../.brief/HARNESS.md import marker to CLAUDE.md");
  } else {
    matrixRow = 5;
    actions.push("Add @../.brief/HARNESS.md import marker to CLAUDE.md");
  }

  actions.unshift("Copy .brief/HARNESS.md");

  return { agentsExists, claudeExists, claudeImportsAgents, matrixRow, actions };
}

export function applyHarness(projectRoot, state) {
  const result = { copied: [], injected: [], skipped: [] };
  const harnessContent = readFileSync(HARNESS_TEMPLATE, "utf8");

  // 1. Copy .brief/HARNESS.md
  copyHarnessFile(projectRoot, harnessContent);
  result.copied.push(".brief/HARNESS.md");

  // 2. Apply matrix row
  const agentsMdPath = join(projectRoot, "AGENTS.md");
  const claudeMdPath = join(projectRoot, ".claude", "CLAUDE.md");

  switch (state.matrixRow) {
    case 1:
      createAgentsWithHarness(agentsMdPath, harnessContent);
      createClaudeWithImport(claudeMdPath);
      result.injected.push("AGENTS.md (created + inline)");
      result.injected.push("CLAUDE.md (created, imports AGENTS.md)");
      break;
    case 2:
      injectInlineMarker(agentsMdPath, harnessContent);
      createClaudeWithImport(claudeMdPath);
      result.injected.push("AGENTS.md (inline marker)");
      result.injected.push("CLAUDE.md (created, imports AGENTS.md)");
      break;
    case 3:
      injectInlineMarker(agentsMdPath, harnessContent);
      result.injected.push("AGENTS.md (inline marker)");
      result.skipped.push("CLAUDE.md (inherits via AGENTS.md import)");
      break;
    case 4:
      injectInlineMarker(agentsMdPath, harnessContent);
      injectImportMarker(claudeMdPath);
      result.injected.push("AGENTS.md (inline marker)");
      result.injected.push("CLAUDE.md (import marker)");
      break;
    case 5:
      injectImportMarker(claudeMdPath);
      result.injected.push("CLAUDE.md (import marker)");
      break;
  }

  return result;
}

// --- Internal functions ---

function copyHarnessFile(projectRoot, content) {
  const briefDir = join(projectRoot, ".brief");
  mkdirSync(briefDir, { recursive: true });
  atomicWrite(join(briefDir, "HARNESS.md"), content);
}

function injectInlineMarker(filePath, harnessContent) {
  const existing = readFileSync(filePath, "utf8");
  const sections = parseMarkers(existing);
  const updated = mergeContent(sections, { harness: "## Brief Workflow\n\n" + harnessContent });
  atomicWrite(filePath, updated);
}

function injectImportMarker(filePath) {
  const existing = readFileSync(filePath, "utf8");
  const sections = parseMarkers(existing);
  const updated = mergeContent(sections, { harness: "@../.brief/HARNESS.md" });
  atomicWrite(filePath, updated);
}

function createAgentsWithHarness(filePath, harnessContent) {
  let skeleton = "";
  if (existsSync(AGENTS_TEMPLATE)) {
    skeleton = readFileSync(AGENTS_TEMPLATE, "utf8") + "\n\n";
  }
  const sections = parseMarkers(skeleton);
  const content = mergeContent(sections, { harness: "## Brief Workflow\n\n" + harnessContent });
  mkdirSync(dirname(filePath), { recursive: true });
  atomicWrite(filePath, content);
}

function createClaudeWithImport(filePath) {
  mkdirSync(dirname(filePath), { recursive: true });
  atomicWrite(filePath, "@../AGENTS.md\n");
}

function atomicWrite(filePath, content) {
  const tmpPath = filePath + ".tmp";
  writeFileSync(tmpPath, content);
  renameSync(tmpPath, filePath);
}
```

Wrap each apply action in try/catch with warning — non-fatal except template missing:
- `atomicWrite` failures → warn and continue
- `parseMarkers` failures → warn and skip that file
- `.brief/` dir creation failure → warn and skip harness copy
