# S-003-T4: Create context lib + command

**Priority:** P1 | **Status:** completed | **Blocked By:** --

## What
Create `lib/context.mjs` (graduated context logic) and `lib/commands/context.mjs` (CLI command routing). Supports `brief context graduate <title>` (via stdin), `brief context list`, and `brief context <query>`.

## Why
Agents discover cross-cutting knowledge during implementation that doesn't fit into story docs. `brief context graduate` gives this knowledge a permanent home in `documentation/context/` with proper frontmatter and slug-based filenames.

## Context
- Context files are stored at `documentation/context/<slug>.md`
- `graduate` reads body from stdin — requires `process.stdin.isTTY` guard (error if interactive)
- Slug: lowercase, replace spaces/special chars with dashes, strip path separators
- Collision handling: append `-2`, `-3`, etc. (don't overwrite)
- `--source=<origin>` flag on graduate (optional, defaults to "manual")
- Frontmatter format: `title`, `created`, `source`
- `list` shows table of all graduated context files
- Search (`brief context <query>`) greps content of context files
- This task has NO dependency on codegraph (T-001) — it's pure file operations

```javascript
import { existsSync, mkdirSync, readFileSync, readdirSync, writeFileSync } from "node:fs";
import { join, basename } from "node:path";

const CONTEXT_DIR = "documentation/context";

function slugify(title) {
  return title
    .toLowerCase()
    .replace(/[^a-z0-9\s-]/g, "")
    .replace(/\s+/g, "-")
    .replace(/-+/g, "-")
    .replace(/^-|-$/g, "");
}

function resolveFilename(dir, slug) {
  const base = `${slug}.md`;
  if (!existsSync(join(dir, base))) return base;
  let i = 2;
  while (existsSync(join(dir, `${slug}-${i}.md`))) i++;
  return `${slug}-${i}.md`;
}

export function graduate(root, title, body, opts = {}) {
  const dir = join(root, CONTEXT_DIR);
  mkdirSync(dir, { recursive: true });

  const slug = slugify(title);
  if (!slug) throw new Error("Title produces empty slug");

  const filename = resolveFilename(dir, slug);
  const source = opts.source || "manual";
  const created = new Date().toISOString().split("T")[0];

  const content = `---
title: ${title}
created: ${created}
source: ${source}
---

${body.trim()}
`;

  const filePath = join(dir, filename);
  writeFileSync(filePath, content);
  return join(CONTEXT_DIR, filename);
}

export function list(root) {
  const dir = join(root, CONTEXT_DIR);
  if (!existsSync(dir)) return [];

  return readdirSync(dir)
    .filter(f => f.endsWith(".md"))
    .map(f => {
      const content = readFileSync(join(dir, f), "utf8");
      const titleMatch = content.match(/^title:\s*(.+)$/m);
      const createdMatch = content.match(/^created:\s*(.+)$/m);
      return {
        file: join(CONTEXT_DIR, f),
        title: titleMatch ? titleMatch[1] : basename(f, ".md"),
        created: createdMatch ? createdMatch[1] : "unknown",
      };
    });
}

export function searchContext(root, query) {
  const dir = join(root, CONTEXT_DIR);
  if (!existsSync(dir)) return [];

  const q = query.toLowerCase();
  return readdirSync(dir)
    .filter(f => f.endsWith(".md"))
    .filter(f => {
      const content = readFileSync(join(dir, f), "utf8").toLowerCase();
      return content.includes(q);
    })
    .map(f => {
      const content = readFileSync(join(dir, f), "utf8");
      const titleMatch = content.match(/^title:\s*(.+)$/m);
      const lines = content.split("\n");
      const matchLine = lines.find(l => l.toLowerCase().includes(q) && !l.startsWith("---"));
      return {
        file: join(CONTEXT_DIR, f),
        title: titleMatch ? titleMatch[1] : basename(f, ".md"),
        match: matchLine ? matchLine.trim() : "",
      };
    });
}
```

**Step 2 -- Create lib/commands/context.mjs:**
- File: `lib/commands/context.mjs` (NEW)

```javascript
import { graduate, list, searchContext } from "../context.mjs";

async function readStdin() {
  const chunks = [];
  for await (const chunk of process.stdin) {
    chunks.push(chunk);
  }
  return Buffer.concat(chunks).toString("utf8");
}

export async function contextCommand(root, args, rawArgs) {
  const subcommand = args[0];

  if (subcommand === "graduate") {
    // TTY guard
    if (process.stdin.isTTY) {
      console.error("Error: body must be piped via stdin.");
      console.error("Usage: echo \"body\" | brief context graduate \"title\" [--source=origin]");
      process.exit(1);
    }

    const title = args[1];
    if (!title) {
      console.error("Usage: echo \"body\" | brief context graduate <title> [--source=origin]");
      process.exit(1);
    }

    const sourceFlag = rawArgs.find(a => a.startsWith("--source="));
    const source = sourceFlag ? sourceFlag.split("=")[1] : "manual";

    const body = await readStdin();
    if (!body.trim()) {
      console.error("Error: stdin body is empty.");
      process.exit(1);
    }

    const path = graduate(root, title, body, { source });
    console.log(`Created: ${path}`);
    return;
  }

  if (subcommand === "list") {
    const items = list(root);
    if (items.length === 0) {
      console.log("No graduated context files.");
      return;
    }
    console.log("Title".padEnd(40) + "Created".padEnd(14) + "Path");
    console.log("-".repeat(80));
    for (const item of items) {
      console.log(
        item.title.slice(0, 38).padEnd(40) +
        item.created.padEnd(14) +
        item.file
      );
    }
    return;
  }

  // Default: search
  if (subcommand) {
    const query = args.join(" ");
    const results = searchContext(root, query);
    if (results.length === 0) {
      console.log("No matching context files.");
      return;
    }
    for (const r of results) {
      console.log(`  ${r.file} — ${r.title}`);
      if (r.match) console.log(`    ${r.match}`);
    }
    return;
  }

  console.error("Usage: brief context <graduate|list|query>");
  console.error("  brief context graduate <title>  (body via stdin)");
  console.error("  brief context list");
  console.error("  brief context <query>");
  process.exit(1);
}
```
