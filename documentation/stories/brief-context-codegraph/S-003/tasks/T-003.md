# S-003-T3: Create query command with all flags

**Priority:** P0 | **Status:** completed | **Blocked By:** S-003-T1

## What
Create `lib/commands/query.mjs` — the `brief query` command that combines code results (from code-index.mjs) with doc results (from search.mjs) into sectioned output. Supports `--callers`, `--impact`, and `--json` flags.

## Why
This is the core user-facing feature. Agents currently choose between `brief search` (docs) and `codegraph explore` (code) depending on what they're looking for. `brief query` eliminates that decision — one command, one answer.

## Context
- `brief query <q>` must return code matches first, then doc matches — sectioned with headers
- `--callers` attaches caller chain to each code match
- `--impact` attaches blast radius to each code match
- `--callers` and `--impact` display is capped at 10 symbols; if more exist, show "N more omitted"
- `--json` returns structured JSON matching the QueryResult schema (uncapped)
- If no `.codegraph/` exists, code section is omitted (not errored)
- `brief search` remains unchanged — doc-only, existing behavior preserved
- Command pattern follows existing convention: `queryCommand(root, args, rawArgs)`

## How

**Step 1 -- Create command file:**
- File: `lib/commands/query.mjs` (NEW)

```javascript
import { getHandle, search, callers as getCallers, impact as getImpact } from "../code-index.mjs";
import { searchFiles } from "../search.mjs";

const DISPLAY_CAP = 10;

export async function queryCommand(root, args, rawArgs) {
  const query = args[0];
  if (!query) {
    console.error("Usage: brief query <query> [--callers] [--impact] [--json]");
    process.exit(1);
  }

  const wantCallers = rawArgs.includes("--callers");
  const wantImpact = rawArgs.includes("--impact");
  const wantJson = rawArgs.includes("--json");

  // Code results
  const handle = getHandle(root);
  let codeResults = [];
  if (handle) {
    const nodes = search(handle, query);
    codeResults = nodes.map(node => {
      const result = {
        name: node.name,
        file: node.file,
        line: node.line,
        kind: node.kind,
      };
      if (wantCallers) {
        result.callers = getCallers(handle, node.id);
      }
      if (wantImpact) {
        result.impact = getImpact(handle, node.id);
      }
      return result;
    });
  }

  // Doc results
  const docResults = await searchFiles(root, query);

  if (wantJson) {
    const output = {
      code: codeResults.map(r => ({
        name: r.name,
        file: r.file,
        line: r.line,
        kind: r.kind,
        ...(r.callers && { callers: r.callers.map(c => ({ name: c.name, file: c.file, line: c.line })) }),
        ...(r.impact && { impact: r.impact.map(i => ({ name: i.name, file: i.file, line: i.line })) }),
      })),
      docs: docResults.map(d => ({
        file: d.file,
        type: d.type || "unknown",
        id: d.id || undefined,
        brief: d.brief || d.match || "",
      })),
    };
    console.log(JSON.stringify(output, null, 2));
    return;
  }

  // Human-readable sectioned output
  if (codeResults.length > 0) {
    console.log("## Code\n");
    for (const r of codeResults) {
      console.log(`  ${r.name} (${r.kind}) — ${r.file}:${r.line}`);
      if (r.callers) {
        const display = r.callers.slice(0, DISPLAY_CAP);
        for (const c of display) {
          console.log(`    ← ${c.name} (${c.file}:${c.line})`);
        }
        if (r.callers.length > DISPLAY_CAP) {
          console.log(`    … ${r.callers.length - DISPLAY_CAP} more omitted`);
        }
      }
      if (r.impact) {
        const display = r.impact.slice(0, DISPLAY_CAP);
        for (const i of display) {
          console.log(`    → ${i.name} (${i.file}:${i.line})`);
        }
        if (r.impact.length > DISPLAY_CAP) {
          console.log(`    … ${r.impact.length - DISPLAY_CAP} more omitted`);
        }
      }
    }
  } else if (handle) {
    console.log("## Code\n\n  No matches.\n");
  }

  if (docResults.length > 0) {
    console.log("\n## Docs\n");
    for (const d of docResults) {
      console.log(`  ${d.file} — ${d.brief || d.match || ""}`);
    }
  } else {
    console.log("\n## Docs\n\n  No matches.\n");
  }
}
```
