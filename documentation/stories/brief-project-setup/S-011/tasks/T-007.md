# S-011-T7: brief doctor — Health Check Command

**Priority:** P1 | **Status:** completed | **Blocked By:** S-011-T4

## What
Create `~/.agents/brief/lib/doctor.mjs` implementing a health check command. Add `doctor` route to `cli.mjs`. The command validates symlinks, checks manifest consistency, detects stale/broken state, and reports structured results. Exits 0 if healthy, 1 if failures. Supports `--json` flag for programmatic consumption.

## Why
After `brief init` sets up a project, things can break: symlinks go stale after a brief package update, manifest drifts from filesystem reality, cloning on a new machine leaves broken links. `brief doctor` is the diagnostic tool — it tells you what's wrong and how to fix it, without modifying anything itself.

## Context
**What doctor checks (from PRD acceptance criteria):**
1. Symlink validation — each manifest-recorded symlink resolves to an existing target
2. Broken link detection — symlinks pointing to missing targets
3. Stale copy detection — if copy mode was used, detect when source is newer than copy
4. Manifest consistency — `.agents/brief-manifest.json` matches actual filesystem
5. Exit code — 0 if all pass, 1 if any fail
6. Structured output — `--json` flag

**Manifest schema (written by T4's init):**
```json
{
  "version": "1.0.0",
  "installed_at": "2026-08-25T10:00:00Z",
  "mode": "symlink",
  "detection": { ... },
  "symlinks": [
    { "source": ".claude/skills/brief", "target": "~/.agents/brief/skills/brief", "status": "ok" }
  ],
  "rules_installed": ["coding-style.md", "backend-standards.md"]
}
```

**Doctor output format (human-readable):**
```
brief doctor
──────────────────────────────────────────────────
Checking project: /Users/dev/my-app

✓ Manifest: valid (.agents/brief-manifest.json)
✓ Symlink: .claude/skills/brief → ~/.agents/brief/skills/brief
✗ Symlink: .agents/skills/brief → (broken: target missing)
✓ Rules: coding-style.md exists
✗ Rules: backend-standards.md missing

1 symlink broken, 1 rule missing
Fix: run `brief init` to repair
──────────────────────────────────────────────────
```

**Doctor output format (--json):**
```json
{
  "healthy": false,
  "checks": [
    { "type": "manifest", "status": "pass" },
    { "type": "symlink", "path": ".claude/skills/brief", "status": "pass" },
    { "type": "symlink", "path": ".agents/skills/brief", "status": "fail", "reason": "broken" },
    { "type": "rule", "file": "coding-style.md", "status": "pass" },
    { "type": "rule", "file": "backend-standards.md", "status": "fail", "reason": "missing" }
  ],
  "summary": { "pass": 3, "fail": 2 }
}
```

**Error cases from TRD exception table:**
- `ManifestMissing`: `brief-manifest.json` not found → "Run `brief init` first." (exit 1)
- `ManifestInvalid`: corrupt JSON → "Manifest corrupted. Delete and re-run init." (exit 1)
- `BrokenSymlink`: target missing → report which link is broken
- `StaleManifest`: manifest records N symlinks but filesystem shows M → report mismatch

## How

**Step 1 -- Create the doctor module:**
- File: `~/.agents/brief/lib/doctor.mjs`

```javascript
import { existsSync, readFileSync, statSync } from "node:fs";
import { join } from "node:path";
import { homedir } from "node:os";
import { verifyLink } from "./symlink.mjs";

export function runDoctor(projectRoot, opts = {}) {
  const { json = false } = opts;
  const checks = [];

  // 1. Manifest existence + validity
  const manifestPath = join(projectRoot, ".agents", "brief-manifest.json");
  if (!existsSync(manifestPath)) {
    if (json) {
      return { healthy: false, error: "manifest-missing", message: "Run `brief init` first." };
    }
    console.error("✗ brief-manifest.json not found. Run `brief init` first.");
    return { healthy: false, exitCode: 1 };
  }

  let manifest;
  try {
    manifest = JSON.parse(readFileSync(manifestPath, "utf8"));
    checks.push({ type: "manifest", status: "pass" });
  } catch {
    if (json) {
      return { healthy: false, error: "manifest-invalid", message: "Delete .agents/brief-manifest.json and re-run init." };
    }
    console.error("✗ Manifest corrupted. Delete .agents/brief-manifest.json and re-run `brief init`.");
    return { healthy: false, exitCode: 1 };
  }

  // 2. Symlink validation
  for (const link of manifest.symlinks || []) {
    const linkPath = join(projectRoot, link.source);
    const result = verifyLink(linkPath, link.target);
    if (result.valid) {
      checks.push({ type: "symlink", path: link.source, status: "pass" });
    } else {
      checks.push({ type: "symlink", path: link.source, status: "fail", reason: result.reason });
    }
  }

  // 3. Rules existence
  for (const ruleFile of manifest.rules_installed || []) {
    const rulePath = join(projectRoot, ".claude", "rules", ruleFile);
    if (existsSync(rulePath)) {
      checks.push({ type: "rule", file: ruleFile, status: "pass" });
    } else {
      checks.push({ type: "rule", file: ruleFile, status: "fail", reason: "missing" });
    }
  }

  // 4. Stale copy detection (if mode is "copy")
  if (manifest.mode === "copy") {
    for (const link of manifest.symlinks || []) {
      const copyPath = join(projectRoot, link.source);
      const sourcePath = link.target.replace(/^~/, homedir());
      if (existsSync(copyPath) && existsSync(sourcePath)) {
        const copyMtime = statSync(copyPath).mtimeMs;
        const sourceMtime = statSync(sourcePath).mtimeMs;
        if (sourceMtime > copyMtime) {
          checks.push({ type: "stale-copy", path: link.source, status: "fail", reason: "source-newer" });
        }
      }
    }
  }

  const pass = checks.filter(c => c.status === "pass").length;
  const fail = checks.filter(c => c.status === "fail").length;
  const healthy = fail === 0;

  return { healthy, checks, summary: { pass, fail }, exitCode: healthy ? 0 : 1 };
}
```

**Step 2 -- Add CLI output formatting:**
```javascript
export function printDoctorResult(projectRoot, result, json) {
  if (json) {
    console.log(JSON.stringify({ healthy: result.healthy, checks: result.checks, summary: result.summary }, null, 2));
    return;
  }

  console.log(`\nbrief doctor`);
  console.log("─".repeat(50));
  console.log(`Checking project: ${projectRoot}\n`);

  for (const check of result.checks) {
    const icon = check.status === "pass" ? "✓" : "✗";
    const label = check.type === "symlink" ? `Symlink: ${check.path}`
      : check.type === "rule" ? `Rules: ${check.file}`
      : check.type === "manifest" ? "Manifest: valid"
      : `${check.type}: ${check.path}`;
    const detail = check.status === "fail" ? ` (${check.reason})` : "";
    console.log(`  ${icon} ${label}${detail}`);
  }

  console.log(`\n${result.summary.pass} passed, ${result.summary.fail} failed`);
  if (!result.healthy) {
    console.log("Fix: run `brief init` to repair");
  }
}
```
